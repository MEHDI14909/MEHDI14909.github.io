export default function App() {
  // Content entrance state
  const [contentVisible, setContentVisible] = useState(false);

  // Email copy indicator
  const [copied, setCopied] = useState(false);
  const copyTimeoutRef = useRef(null);

  // "Explore Capabilities" interactive modal state
  const [capabilitiesModalOpen, setCapabilitiesModalOpen] = useState(false);
  const [selectedTech, setSelectedTech] = useState('react');
  const [expandedQuestion, setExpandedQuestion] = useState(0);

  // "Extra Capabilities" interactive modal state (PHP, .NET, C#, SQL Server, MySQL)
  const [extraModalOpen, setExtraModalOpen] = useState(false);
  const [selectedExtraSkill, setSelectedExtraSkill] = useState('php');
  const [expandedExtraQuestion, setExpandedExtraQuestion] = useState(0);

  // "Open" high-level capabilities modal state
  const [openModalActive, setOpenModalActive] = useState(false);
  const [highLevelCategory, setHighLevelCategory] = useState('frontend');
  const [expandedHighLevelQuestion, setExpandedHighLevelQuestion] = useState(0);

  // "Let's Talk" contact modal state
  const [contactModalOpen, setContactModalOpen] = useState(false);
  const [senderName, setSenderName] = useState('');
  const [senderEmail, setSenderEmail] = useState('');
  const [senderMessage, setSenderMessage] = useState('');
  const [messageSent, setMessageSent] = useState(false);

  // Top level container reference for dynamic background illumination
  const containerRef = useRef(null);

  // Canvas frame sequence references
  const canvasRef = useRef(null);
  const videoFallbackRef = useRef(null);
  const framesRef = useRef([]);
  const [framesLoaded, setFramesLoaded] = useState(false);
  const lastDrawnFrameRef = useRef(-1);
  const ctxRef = useRef(null);

  // Mouse tracking references for smooth cursor tracking
  const targetMouseRef = useRef({ x: 0.5, y: 0.5 });
  const currentMouseRef = useRef({ x: 0.5, y: 0.5 });
  const stageRef = useRef(null);
  const rafIdRef = useRef(null);

  // Preload all 96 extracted frames for instantaneous, zero-latency 60fps tracking
  useEffect(() => {
    let isCancelled = false;
    const TOTAL_FRAMES = 96;
    const images = [];
    let loadedCount = 0;

    for (let i = 1; i <= TOTAL_FRAMES; i++) {
      const img = new Image();
      const numStr = String(i).padStart(3, '0');
      img.src = `/assets/character_frames/f_${numStr}.jpg`;
      img.onload = () => {
        if (isCancelled) return;
        loadedCount++;
        if (loadedCount >= 8 || i === 50) {
          setFramesLoaded(true);
        }
      };
      img.onerror = () => {
        if (i === 50 && !isCancelled) {
          setFramesLoaded(true);
        }
      };
      images.push(img);
    }
    framesRef.current = images;

    return () => {
      isCancelled = true;
    };
  }, []);

  // Entrance animation trigger
  useEffect(() => {
    const timer = window.setTimeout(() => {
      setContentVisible(true);
    }, 100);
    return () => window.clearTimeout(timer);
  }, []);

  // Track cursor and touch events
  useEffect(() => {
    const handleMouseMove = (e) => {
      const normX = Math.min(Math.max(0, e.clientX / window.innerWidth), 1);
      const normY = Math.min(Math.max(0, e.clientY / window.innerHeight), 1);
      targetMouseRef.current = { x: normX, y: normY };
    };

    const handleMouseLeave = () => {
      targetMouseRef.current = { x: 0.5, y: 0.5 };
    };

    const handleTouchMove = (e) => {
      if (e.touches.length === 0) return;
      const touch = e.touches[0];
      const normX = Math.min(Math.max(0, touch.clientX / window.innerWidth), 1);
      const normY = Math.min(Math.max(0, touch.clientY / window.innerHeight), 1);
      targetMouseRef.current = { x: normX, y: normY };
    };

    const handleTouchEnd = () => {
      targetMouseRef.current = { x: 0.5, y: 0.5 };
    };

    window.addEventListener('mousemove', handleMouseMove);
    window.addEventListener('mouseleave', handleMouseLeave);
    window.addEventListener('touchmove', handleTouchMove, { passive: true });
    window.addEventListener('touchend', handleTouchEnd);

    return () => {
      window.removeEventListener('mousemove', handleMouseMove);
      window.removeEventListener('mouseleave', handleMouseLeave);
      window.removeEventListener('touchmove', handleTouchMove);
      window.removeEventListener('touchend', handleTouchEnd);
    };
  }, []);

  // 60FPS Animation Loop: Optimized for responsiveness, fluid gaze tracking, and zero lag
  useEffect(() => {
    const LERP_FACTOR = 0.085;
    let isRunning = true;

    const update = () => {
      if (!isRunning) return;

      currentMouseRef.current.x +=
        (targetMouseRef.current.x - currentMouseRef.current.x) * LERP_FACTOR;
      currentMouseRef.current.y +=
        (targetMouseRef.current.y - currentMouseRef.current.y) * LERP_FACTOR;

      const currX = currentMouseRef.current.x;

      // CORRECTED CURSOR TRACKING:
      // Frame 28 = looks RIGHT, Frame 50 = looks CENTER, Frame 72 = looks LEFT
      const targetFrame = Math.round(50 + (0.5 - currX) * 44);
      const clampedFrame = Math.max(26, Math.min(74, targetFrame));

      // Instant 0ms Canvas Rendering: Pure character render with full shoes & natural ground shadow
      if (clampedFrame !== lastDrawnFrameRef.current) {
        const canvas = canvasRef.current;
        const frames = framesRef.current;
        if (canvas && frames.length >= clampedFrame) {
          const img = frames[clampedFrame - 1];
          if (img && img.complete && img.naturalWidth > 0) {
            if (!ctxRef.current) {
              ctxRef.current = canvas.getContext('2d');
            }
            const ctx = ctxRef.current;
            if (ctx) {
              // Draw complete image without any destructive rectangular cuts over shoes
              ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
              lastDrawnFrameRef.current = clampedFrame;
            }
          }
        }
      }

      // Fallback video sync
      const video = videoFallbackRef.current;
      if (video && video.duration && !video.seeking) {
        const targetVideoTime = (clampedFrame / 96) * video.duration;
        if (Math.abs(video.currentTime - targetVideoTime) > 0.03) {
          video.currentTime = targetVideoTime;
        }
      }

      // Character remains completely stable without shaking or tilting ("Hile na")!
      rafIdRef.current = requestAnimationFrame(update);
    };

    rafIdRef.current = requestAnimationFrame(update);

    return () => {
      isRunning = false;
      if (rafIdRef.current) {
        cancelAnimationFrame(rafIdRef.current);
      }
    };
  }, []);

  // Copy email handler
  const handleCopyEmail = (e) => {
    if (e) {
      e.preventDefault();
      e.stopPropagation();
    }
    navigator.clipboard
      .writeText('syedmehdi.dev@gmail.com')
      .then(() => {
        setCopied(true);
        if (copyTimeoutRef.current) {
          window.clearTimeout(copyTimeoutRef.current);
        }
        copyTimeoutRef.current = window.setTimeout(() => {
          setCopied(false);
        }, 2200);
      })
      .catch((err) => {
        console.error('Failed to copy to clipboard', err);
      });
  };

  const handleOpenTechCapabilities = (techKey) => {
    setSelectedTech(techKey);
    setExpandedQuestion(0);
    setCapabilitiesModalOpen(true);
  };

  const handleSendContactEmail = (e) => {
    e.preventDefault();
    const mailtoSubject = encodeURIComponent(
      `Project Inquiry from ${senderName || 'Client'} (${senderEmail || 'Portfolio'})`
    );
    const mailtoBody = encodeURIComponent(
      `Hello Syed Mehdi,\n\n${senderMessage}\n\n---\nSender Name: ${senderName}\nSender Email: ${senderEmail}`
    );
    const mailtoUrl = `mailto:syedmehdi.dev@gmail.com?subject=${mailtoSubject}&body=${mailtoBody}`;
    window.open(mailtoUrl, '_blank');
    setMessageSent(true);
    setTimeout(() => {
      setMessageSent(false);
      setSenderMessage('');
    }, 4000);
  };

  // Top Dock navigation items
  const dockNavItems = [
    {
      icon: Home,
      label: 'Home',
      onClick: () => {
        window.scrollTo({ top: 0, behavior: 'smooth' });
        setCapabilitiesModalOpen(false);
        setOpenModalActive(false);
        setExtraModalOpen(false);
        setContactModalOpen(false);
      },
    },
    {
      icon: Briefcase,
      label: 'Capabilities',
      onClick: () => handleOpenTechCapabilities('react'),
    },
    {
      icon: Code2,
      label: 'Frontend',
      onClick: () => handleOpenTechCapabilities('react'),
    },
    {
      icon: Server,
      label: 'Backend',
      onClick: () => handleOpenTechCapabilities('nodejs'),
    },
    {
      icon: Mail,
      label: 'Contact',
      onClick: () => setContactModalOpen(true),
    },
  ];

  return (
    <div
      ref={containerRef}
      className="relative w-screen h-screen overflow-hidden select-none text-white bg-[#A2090F]"
      style={{
        backgroundColor: '#A2090F',
      }}
    >
      {/* Interactive Spotlight Cursor: Smooth cursor light effect */}
      <SpotlightCursor
        config={{
          radius: 260,
          brightness: 0.14,
          color: '#ffffff',
        }}
      />

      {/* 3D CHARACTER STAGE: Restored to right side ("barabar mein") for laptop, tablet, and mobile */}
      <div className="fixed inset-y-0 right-0 w-[52%] sm:w-[54%] md:w-[58%] lg:w-[56%] xl:w-[52%] h-screen z-0 pointer-events-none overflow-hidden flex items-end justify-end pb-0 sm:pb-1 md:pb-2 lg:pb-3">
        <div
          ref={stageRef}
          className="relative w-full h-full flex items-end justify-end pr-0 sm:pr-2 md:pr-6 lg:pr-12 will-change-transform pb-0"
        >
          {/* 60FPS Frame-Sequence Canvas: Scaled slightly larger & perfectly grounded */}
          <canvas
            ref={canvasRef}
            width={1280}
            height={720}
            style={{
              WebkitMaskImage:
                'linear-gradient(to right, transparent 0%, rgba(0,0,0,0.3) 3%, black 10%, black 100%)',
              maskImage:
                'linear-gradient(to right, transparent 0%, rgba(0,0,0,0.3) 3%, black 10%, black 100%)',
            }}
            className={`w-full h-auto max-h-[95vh] sm:max-h-[96vh] md:max-h-[98vh] scale-116 sm:scale-124 md:scale-134 lg:scale-140 xl:scale-145 origin-bottom-right object-contain md:object-cover object-bottom select-none pointer-events-none transition-opacity duration-300 ${
              framesLoaded ? 'opacity-100' : 'opacity-0'
            }`}
          />

          {/* Fallback Video if frames loading */}
          {!framesLoaded && (
            <video
              ref={videoFallbackRef}
              src="/assets/aistudio/Create_left_and_right_image_b25dda15_8d2c20fc.mp4"
              muted
              playsInline
              preload="auto"
              style={{
                WebkitMaskImage:
                  'linear-gradient(to right, transparent 0%, rgba(0,0,0,0.3) 3%, black 10%, black 100%)',
                maskImage:
                  'linear-gradient(to right, transparent 0%, rgba(0,0,0,0.3) 3%, black 10%, black 100%)',
              }}
              className="absolute inset-0 w-full h-full object-contain md:object-cover object-bottom pointer-events-none select-none scale-116 sm:scale-124 md:scale-134 lg:scale-140 xl:scale-145 origin-bottom-right"
            />
          )}
        </div>
      </div>

      {/* TOP NAVIGATION AREA: Clean logo space on left + Dock navbar centered + Let's Talk button on right */}
      <header className="fixed top-0 left-0 right-0 z-30 w-full px-3 sm:px-6 md:px-10 lg:px-14 py-2.5 sm:py-3.5 md:py-5 flex items-center justify-between pointer-events-none">

                  onClick={() => {
                    setExtraModalOpen(false);
                    setContactModalOpen(true);
                  }}
                  className="grow sm:grow-0 inline-flex items-center justify-center gap-2 px-5 py-2 rounded-full text-xs font-bold bg-gradient-to-r from-[#A0080E] to-[#EA580C] hover:from-[#8B080E] hover:to-[#F97316] text-white shadow-md hover:shadow-[0_0_15px_rgba(234,88,12,0.4)] transition-all cursor-pointer"
                >
                  <Mail className="w-3.5 h-3.5" />
                  <span>Send Email</span>
                </button>
              </div>
            </div>
          </div>
        </div>
      )}

      {/* "OPEN" BUTTON MODAL (High-level category questions: Frontend, Backend, Full Stack) */}
      {openModalActive && (
        <div
          role="dialog"
          aria-modal="true"
          className="fixed inset-0 z-50 flex items-center justify-center p-3 sm:p-6 bg-black/80 backdrop-blur-md animate-fade-in"
          onClick={() => setOpenModalActive(false)}
        >
          <div
            className="bg-[#121214] text-white rounded-3xl max-w-2xl w-full max-h-[88vh] flex flex-col shadow-2xl relative border border-white/20 overflow-hidden"
            onClick={(e) => e.stopPropagation()}
            style={{ fontFamily: 'var(--font-body)' }}
          >
            <div className="h-1.5 w-full bg-gradient-to-r from-[#A0080E] via-[#EA580C] to-amber-400 shrink-0" />

            <div className="px-6 sm:px-8 pt-6 pb-4 border-b border-white/10 flex items-start justify-between shrink-0">
              <div>
                <div className="flex items-center gap-2 mb-1.5 text-xs font-bold tracking-widest uppercase text-orange-400">
                  <Sparkles className="w-3.5 h-3.5 text-orange-400" />
                  <span>Developer Capabilities & Q&A</span>
                </div>
                <h3
                  className="text-2xl sm:text-3xl font-extrabold tracking-tight text-white"
                  style={{ fontFamily: 'var(--font-heading)' }}
                >
                  What I Can Build For You
                </h3>
                <p className="text-xs sm:text-sm text-neutral-400 mt-1">
                  Select a category below to explore specific questions and realistic development capabilities.
                </p>
              </div>

              <button
                type="button"
                onClick={() => setOpenModalActive(false)}
                className="text-neutral-400 hover:text-white hover:bg-white/10 rounded-full p-2 transition-colors cursor-pointer shrink-0 ml-4"
                aria-label="Close dialog"
              >
                <X className="w-5 h-5" />
              </button>
            </div>

            {/* 3 Categories: Frontend, Backend, Full Stack */}
            <div className="px-6 sm:px-8 py-3.5 bg-black/40 border-b border-white/10 shrink-0">
              <div className="grid grid-cols-3 gap-2">
                {[
                  { key: 'frontend', label: 'FRONTEND DEVELOPER', icon: Monitor },
                  { key: 'backend', label: 'BACKEND DEVELOPER', icon: Server },
                  { key: 'fullstack', label: 'FULL STACK DEVELOPER', icon: Layers3 },
                ].map((tab) => {
                  const isActive = highLevelCategory === tab.key;
                  const Icon = tab.icon;
                  return (
                    <button
                      key={tab.key}
                      type="button"
                      onClick={() => {
                        setHighLevelCategory(tab.key);
                        setExpandedHighLevelQuestion(0);
                      }}
                      className={`flex flex-col sm:flex-row items-center justify-center gap-1.5 sm:gap-2 px-2.5 py-2.5 rounded-xl text-[11px] sm:text-xs font-bold tracking-wider transition-all duration-250 cursor-pointer text-center ${
                        isActive
                          ? 'bg-gradient-to-r from-[#A0080E] to-[#EA580C] text-white shadow-[0_0_15px_rgba(234,88,12,0.4)] border border-orange-400/60'
                          : 'bg-white/5 hover:bg-white/10 text-neutral-300 hover:text-white border border-white/5'
                      }`}
                    >
                      <Icon className={`w-4 h-4 shrink-0 ${isActive ? 'text-white' : 'text-neutral-400'}`} />
                      <span className="leading-tight">{tab.label}</span>
                    </button>
                  );
                })}
              </div>
            </div>

            {/* High Level Questions Body */}
            <div className="p-6 sm:p-8 overflow-y-auto space-y-4 grow">
              <div className="p-4 rounded-2xl bg-white/5 border border-white/10 flex items-start gap-3.5">
                {highLevelCategory === 'frontend' && <Monitor className="w-5 h-5 text-orange-400 shrink-0 mt-0.5" />}
                {highLevelCategory === 'backend' && <Server className="w-5 h-5 text-orange-400 shrink-0 mt-0.5" />}
                {highLevelCategory === 'fullstack' && <Layers3 className="w-5 h-5 text-orange-400 shrink-0 mt-0.5" />}
                <div>
                  <h4 className="text-sm font-bold text-white tracking-wide uppercase">
                    {highLevelCategory === 'frontend' && 'Frontend Development'}
                    {highLevelCategory === 'backend' && 'Backend Development'}
                    {highLevelCategory === 'fullstack' && 'Full Stack Development'}
                  </h4>
                  <p className="text-xs sm:text-sm text-neutral-300 leading-relaxed mt-0.5">
                    {highLevelCategory === 'frontend' &&
                      'Building responsive, interactive, and component-based user interfaces with React.js, modern JavaScript, semantic HTML, and responsive CSS styling.'}
                    {highLevelCategory === 'backend' &&
                      'Engineering reliable backend services, RESTful API endpoints, server routing, and database connectivity with Node.js and Express.js.'}
                    {highLevelCategory === 'fullstack' &&
                      'Connecting frontend interfaces with backend services, integrating REST APIs, and managing application data flow across the entire stack.'}
                  </p>
                </div>
              </div>

              {/* Sample high level questions */}
              <div className="space-y-3 pt-1">
                {(highLevelCategory === 'frontend'
                  ? [
                      {
                        q: 'What can you build with React.js and JavaScript?',
                        a: 'I build clean, responsive, single-page web applications and interactive interfaces using React.js and modern JavaScript. I break down interfaces into reusable functional components, handle component lifecycle and state with React hooks (useState, useEffect, useMemo), and ensure fast, predictable UI rendering.',
                        t: ['React.js', 'JavaScript (ES6+)', 'Component Architecture'],
                      },
                      {
                        q: 'How do you handle HTML & CSS styling and responsive layouts?',
                        a: 'I write semantic HTML5 for clean structure and accessibility. For styling, I utilize modern CSS (Flexbox, CSS Grid) along with utility frameworks like Tailwind CSS to build mobile-first responsive layouts that look great on smartphones, tablets, laptops, and wide monitors.',
                        t: ['HTML5', 'CSS3', 'Tailwind CSS', 'Responsive Layouts'],
                      },
                    ]
                  : highLevelCategory === 'backend'
                  ? [
                      {
                        q: 'What backend services can you develop with Node.js & Express.js?',
                        a: 'I develop server-side applications using Node.js and the Express.js framework. I configure middleware, set up structured route handlers for different resources, process incoming JSON payloads, handle CORS, and secure endpoints with proper HTTP status codes.',
                        t: ['Node.js', 'Express.js', 'Server Routing', 'Middleware'],
                      },
                      {
                        q: 'How do you manage database connectivity and execute queries?',
                        a: 'I connect Node.js backends to databases (such as PostgreSQL, MySQL, and document databases). I design tables, define relational schemas, write efficient queries, handle filtering and sorting, and ensure database connections are pooled and closed safely.',
                        t: ['Database Connectivity', 'SQL Queries', 'Data Persistence'],
                      },
                    ]
                  : [
                      {
                        q: 'How do you integrate frontend interfaces with backend services?',
                        a: 'I connect React client components to Node.js/Express backend APIs using asynchronous fetch/axios requests. I manage request loading states, display clear validation and error messages to users, and update UI state in real time upon successful server responses.',
                        t: ['Frontend & Backend Integration', 'Async Data Fetching', 'State Sync'],
                      },
                      {
                        q: 'What is your process for complete web application development?',
                        a: 'I take a project from concept to working product: starting with data modeling and database setup, building and testing the Express API routes, crafting the responsive React frontend, and linking the two together so users experience a cohesive, working application.',
                        t: ['End-to-End Development', 'Architecture Flow', 'Functional Web Apps'],
                      },
                    ]
                ).map((item, idx) => {
                  const isExp = expandedHighLevelQuestion === idx;
                  return (
                    <div
                      key={idx}
                      className={`rounded-2xl border transition-all duration-250 overflow-hidden ${
                        isExp
                          ? 'bg-[#1a1a1e] border-orange-500/40 shadow-md'
                          : 'bg-white/5 border-white/10 hover:border-white/20'
                      }`}
                    >
                      <button
                        type="button"
                        onClick={() => setExpandedHighLevelQuestion(isExp ? null : idx)}
                        className="w-full text-left px-5 py-4 flex items-center justify-between gap-3 cursor-pointer group"
                      >
                        <span className="text-sm sm:text-base font-bold text-white group-hover:text-orange-200 transition-colors">
                          {item.q}
                        </span>
                        <ChevronDown
                          className={`w-4 h-4 text-orange-400 shrink-0 transition-transform duration-250 ${
                            isExp ? 'rotate-180 text-orange-300' : ''
                          }`}
                        />
                      </button>
                      {isExp && (
                        <div className="px-5 pb-5 pt-1 border-t border-white/5">
                          <p className="text-xs sm:text-sm text-neutral-300 leading-relaxed font-normal mb-3">
                            {item.a}
                          </p>
                          <div className="flex flex-wrap items-center gap-1.5">
                            {item.t.map((tg) => (
                              <span
                                key={tg}
                                className="text-[10px] sm:text-[11px] font-bold px-2.5 py-0.5 rounded-full bg-orange-500/15 text-orange-300 border border-orange-500/30"
                              >
                                {tg}
                              </span>
                            ))}
                          </div>
                        </div>
                      )}
                    </div>
                  );
                })}
              </div>
            </div>

            <div className="px-6 sm:px-8 py-4 border-t border-white/10 bg-black/40 flex flex-col sm:flex-row items-center justify-between gap-3 shrink-0">
              <span className="text-xs text-neutral-400 text-center sm:text-left">
                Ready to discuss your project? Reach out directly via email.
              </span>
              <div className="flex items-center gap-2.5 w-full sm:w-auto">
                <button
                  type="button"
                  onClick={handleCopyEmail}
                  className="grow sm:grow-0 inline-flex items-center justify-center gap-2 px-4 py-2 rounded-full text-xs font-bold border border-white/20 text-neutral-200 hover:text-white hover:bg-white/10 transition-colors cursor-pointer"
                >
                  {copied ? <Check className="w-3.5 h-3.5 text-emerald-400" /> : <Copy className="w-3.5 h-3.5" />}
                  <span>{copied ? 'Copied' : 'Copy Email'}</span>
                </button>
                <button
                  type="button"
                  onClick={() => {
                    setOpenModalActive(false);
                    setContactModalOpen(true);
                  }}
                  className="grow sm:grow-0 inline-flex items-center justify-center gap-2 px-5 py-2 rounded-full text-xs font-bold bg-gradient-to-r from-[#A0080E] to-[#EA580C] hover:from-[#8B080E] hover:to-[#F97316] text-white shadow-md hover:shadow-[0_0_15px_rgba(234,88,12,0.4)] transition-all cursor-pointer"
                >
                  <Mail className="w-3.5 h-3.5" />
                  <span>Send Email</span>
                </button>
              </div>
            </div>
          </div>
        </div>
      )}

      {/* "LET'S TALK" INTERACTIVE CONTACT MODAL (WITH FULL EDITABLE FORM & SEND BUTTON) */}
      {contactModalOpen && (
        <div
          role="dialog"
          aria-modal="true"
          className="fixed inset-0 z-50 flex items-center justify-center p-3 sm:p-5 bg-black/80 backdrop-blur-md animate-fade-in"
          onClick={() => setContactModalOpen(false)}
        >
          <div
            className="bg-[#121214] text-white rounded-3xl max-w-lg w-full max-h-[92vh] flex flex-col p-6 sm:p-7 shadow-2xl relative border border-white/20 overflow-hidden"
            onClick={(e) => e.stopPropagation()}
            style={{ fontFamily: 'var(--font-body)' }}
          >
            {/* Top Accent Line */}
            <div className="h-1.5 absolute top-0 left-0 right-0 bg-gradient-to-r from-[#A0080E] via-[#EA580C] to-amber-400" />

            {/* Close Button */}
            <button
              type="button"
              onClick={() => setContactModalOpen(false)}
              className="absolute top-5 right-5 text-neutral-400 hover:text-white hover:bg-white/10 rounded-full p-2 transition-colors cursor-pointer"
              aria-label="Close dialog"
            >
              <X className="w-5 h-5" />
            </button>

            {/* Header */}
            <div className="flex items-center gap-2 mb-1 text-xs font-bold tracking-widest uppercase text-orange-400">
              <Mail className="w-3.5 h-3.5 text-orange-400" />
              <span>Contact Syed Mehdi</span>
            </div>

            <h3
              className="text-2xl sm:text-3xl font-extrabold tracking-tight mb-1 text-white"
              style={{ fontFamily: 'var(--font-heading)' }}
            >
              Let's Connect
            </h3>

            <p className="text-xs sm:text-sm font-normal text-neutral-400 mb-4">
              Write your email and message below, and click Send Email to get in touch directly.
            </p>

            {/* Direct Email Card with 1-Click Copy */}
            <div className="p-3 rounded-xl bg-white/5 border border-white/10 flex items-center justify-between mb-4">
              <div>
                <span className="text-[10px] font-bold text-neutral-400 uppercase tracking-wider block">
                  Direct Email Address
                </span>
                <span className="text-xs sm:text-sm font-bold text-white select-all">
                  syedmehdi.dev@gmail.com
                </span>
              </div>
              <button
                type="button"
                onClick={handleCopyEmail}
                className="px-3 py-1.5 rounded-lg bg-white/10 hover:bg-white/20 text-xs font-semibold text-white transition-colors cursor-pointer"
              >
                {copied ? 'Copied!' : 'Copy'}
              </button>
            </div>

            {/* Full Editable Form with Inputs & Send Button */}
            <form onSubmit={handleSendContactEmail} className="space-y-3 grow overflow-y-auto pr-0.5">
              <div>
                <label className="block text-[11px] font-bold text-white/80 uppercase tracking-wider mb-1">
                  Your Name
                </label>
                <input
                  type="text"
                  required
                  value={senderName}
                  onChange={(e) => setSenderName(e.target.value)}
                  placeholder="Enter your name"
                  className="w-full px-3.5 py-2 rounded-xl bg-white/5 border border-white/20 text-white placeholder-neutral-500 text-xs sm:text-sm focus:outline-none focus:border-orange-400 focus:bg-white/10 transition-colors"
                />
              </div>

              <div>
                <label className="block text-[11px] font-bold text-white/80 uppercase tracking-wider mb-1">
                  Your Email
                </label>
                <input
                  type="email"
                  required
                  value={senderEmail}
                  onChange={(e) => setSenderEmail(e.target.value)}
                  placeholder="your.email@example.com"
                  className="w-full px-3.5 py-2 rounded-xl bg-white/5 border border-white/20 text-white placeholder-neutral-500 text-xs sm:text-sm focus:outline-none focus:border-orange-400 focus:bg-white/10 transition-colors"
                />
              </div>

              <div>
                <label className="block text-[11px] font-bold text-white/80 uppercase tracking-wider mb-1">
                  Message / Project Details
                </label>
                <textarea
                  required
                  rows={3}
                  value={senderMessage}
                  onChange={(e) => setSenderMessage(e.target.value)}
                  placeholder="Tell me what you'd like to build (React app, full stack system, API integration, etc.)..."
                  className="w-full px-3.5 py-2 rounded-xl bg-white/5 border border-white/20 text-white placeholder-neutral-500 text-xs sm:text-sm focus:outline-none focus:border-orange-400 focus:bg-white/10 transition-colors resize-none"
                />
              </div>

              {messageSent && (
                <div className="p-2.5 rounded-xl bg-emerald-500/20 border border-emerald-400/40 text-emerald-300 text-xs font-semibold flex items-center gap-2">
                  <Check className="w-4 h-4 shrink-0" />
                  <span>Opening your email client to send to syedmehdi.dev@gmail.com!</span>
                </div>
              )}

              <div className="flex items-center gap-3 pt-2">
                <button
                  type="submit"
                  className="grow inline-flex items-center justify-center gap-2 bg-gradient-to-r from-[#A0080E] to-[#EA580C] hover:from-[#8B080E] hover:to-[#F97316] text-white py-2.5 rounded-full text-xs sm:text-sm font-bold shadow-md hover:shadow-[0_0_18px_rgba(234,88,12,0.45)] transition-all cursor-pointer"
                >
                  <Send className="w-4 h-4" />
                  <span>Send Email</span>
                </button>
                <button
                  type="button"
                  onClick={() => setContactModalOpen(false)}
                  className="px-5 py-2.5 rounded-full text-xs sm:text-sm font-semibold border border-white/20 text-neutral-300 hover:text-white hover:bg-white/10 transition-colors cursor-pointer"
                >
                  Close
                </button>
              </div>
            </form>
          </div>
        </div>
      )}
    </div>
  );
}
